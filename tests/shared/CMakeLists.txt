# Standalone shared-library checks, without configuring either desktop product.
cmake_minimum_required(VERSION 3.21)
project(EightBitSharedChecks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
add_compile_definitions(SCI_OWNREGEX)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Core5Compat Test)
qt_standard_project_setup()
enable_testing()
get_filename_component(EDITOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(dependency lua scintilla lexilla uchardet)
    add_subdirectory("${EDITOR_ROOT}/thirdparty/${dependency}" "thirdparty/${dependency}" EXCLUDE_FROM_ALL)
endforeach()
target_link_libraries(lexilla PRIVATE scintilla)
add_subdirectory("${EDITOR_ROOT}/src/editor" editor EXCLUDE_FROM_ALL)
foreach(suite EditorLibrary ThemePresentation VimMode)
    qt_add_executable(${suite}Test "${EDITOR_ROOT}/tests/${suite}Test.cpp")
    target_link_libraries(${suite}Test PRIVATE EightBit::Editor Qt6::Test)
    add_test(NAME ${suite} COMMAND ${suite}Test)
    set_tests_properties(${suite} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endforeach()
